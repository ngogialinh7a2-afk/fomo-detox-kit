            const tasks = [

                "Để điện thoại xa giường ngủ ít nhất 2 mét",

            let html = "";
            for(let i = 1; i <= days; i++) {

                        <div class="flex items-center gap-3">

        // --- MODULE 4: WEB AUDIO API SYNTHESIZER ---
        function initAudioContext() {
            if(!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }

        function stopAudio() {
            if(currentOsc) {
                currentOsc.stop();
                currentOsc.disconnect();
                currentOsc = null;
            }
            isAudioPlaying = false;
            document.getElementById('master-play-btn').innerHTML = `<i class="fa-solid fa-play"></i>`;
            document.getElementById('current-sound-title').innerText = "Đã Dừng Phát Âm Thanh";
            document.getElementById('current-sound-desc').innerText = "Chọn dải âm thanh để hỗ trợ giấc ngủ.";
        }

        function playPreset(type) {
            initAudioContext();
            stopAudio();

            currentGain = audioCtx.createGain();
            currentGain.gain.setValueAtTime(0.15, audioCtx.currentTime);
            currentGain.connect(audioCtx.destination);

            if(type === 'alpha') {
                // Alpha wave synthesizer 10Hz binaural simulation tone
                currentOsc = audioCtx.createOscillator();
                currentOsc.type = 'sine';
                currentOsc.frequency.setValueAtTime(210, audioCtx.currentTime); // Base tone
                currentOsc.connect(currentGain);
                currentOsc.start();

                document.getElementById('current-sound-title').innerText = "Đang Phát: Sóng Alpha Thư Giãn (10Hz)";
                document.getElementById('current-sound-desc').innerText = "Dải tần xoa dịu thần kinh, làm giảm lo âu và căng thẳng.";
            } else if(type === 'theta') {
                // Theta wave synthesizer 6Hz simulation tone
                currentOsc = audioCtx.createOscillator();
                currentOsc.type = 'triangle';
                currentOsc.frequency.setValueAtTime(136.1, audioCtx.currentTime); // Om frequency
                currentOsc.connect(currentGain);
                currentOsc.start();

                document.getElementById('current-sound-title').innerText = "Đang Phát: Sóng Theta Đi Vào Giấc Ngủ (6Hz)";
                document.getElementById('current-sound-desc').innerText = "Đưa não bộ vào trạng thái ngủ sâu và hồi phục nhịp sinh học.";
            } else if(type === 'rain') {
                // White/Pink noise simulation using buffer
                const bufferSize = audioCtx.sampleRate * 2;
                const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
                const data = buffer.getChannelData(0);
                let b0 = 0, b1 = 0, b2 = 0, b3 = 0, b4 = 0, b5 = 0, b6 = 0;
                for (let i = 0; i < bufferSize; i++) {
                    let white = Math.random() * 2 - 1;
                    b0 = 0.99886 * b0 + white * 0.0555179;
                    b1 = 0.99332 * b1 + white * 0.0750759;
                    b2 = 0.96900 * b2 + white * 0.1538520;
                    b3 = 0.86650 * b3 + white * 0.3104856;
                    b4 = 0.55000 * b4 + white * 0.5329522;
                    b5 = -0.7616 * b5 - white * 0.0168980;
                    data[i] = b0 + b1 + b2 + b3 + b4 + b5 + b6 + white * 0.5362;
                    data[i] *= 0.11; // volume adjust
                    b6 = white * 0.115926;
                }
                const noise = audioCtx.createBufferSource();
                noise.buffer = buffer;
                noise.loop = true;
                noise.connect(currentGain);
                noise.start();
                currentOsc = noise;

                document.getElementById('current-sound-title').innerText = "Đang Phát: Tiếng Mưa Thư Giãn (Pink Noise)";
                document.getElementById('current-sound-desc').innerText = "Tiếng mưa rơi êm dịu giúp lấn át tiếng ồn môi trường.";
            } else if(type === 'zen') {
                // Zen Lullaby chord synthesis
                currentOsc = audioCtx.createOscillator();
                currentOsc.type = 'sine';
                currentOsc.frequency.setValueAtTime(174, audioCtx.currentTime); // Solfeggio 174Hz
                currentOsc.connect(currentGain);
                currentOsc.start();

                document.getElementById('current-sound-title').innerText = "Đang Phát: Nhạc Thiền Êm Dịu (<60 BPM)";
                document.getElementById('current-sound-desc').innerText = "Giai điệu chậm rãi xoa dịu nhịp tim trước giờ ngủ.";
            }

            isAudioPlaying = true;
            document.getElementById('master-play-btn').innerHTML = `<i class="fa-solid fa-pause"></i>`;
        }

        function toggleGlobalAudio() {
            if(isAudioPlaying) {
                stopAudio();
            } else {
                playPreset('alpha');
            }
        }

        function setSleepTimer(minutes) {
            if(sleepTimerTimeout) clearTimeout(sleepTimerTimeout);
            const mins = parseInt(minutes);
            if(mins > 0) {
                sleepTimerTimeout = setTimeout(() => {
                    stopAudio();
                    alert(`⏰ Đã hết thời gian hẹn giờ (${mins} phút). Chúc bạn có một giấc ngủ ngon! 🌙`);
                }, mins * 60 * 1000);
            }
        }

        function renderConfessions() {
            const feed = document.getElementById('confessions-feed');
            feed.innerHTML = confessions.map(c => `

        function addConfession() {
            const input = document.getElementById('confession-input');
            let text = input.value.trim();
            if(!text) return;

            // Bad word auto filter
            badWords.forEach(word => {
                const reg = new RegExp(word, 'gi');
                text = text.replace(reg, '***');
            });

            confessions.unshift({
                id: Date.now(),
                text: text,
                likes: 1,
                time: "Vừa xong"
            });

            input.value = "";
            renderConfessions();
        }

        function likeConfession(id) {
            const item = confessions.find(c => c.id === id);
            if(item) {
                item.likes++;
                renderConfessions();
            }
        }
